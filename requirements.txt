# Core pipeline + app - pinned exactly (not >=) so CI, local dev, and a fresh clone all resolve
# to the same versions this project was built and tested against. Verified on Python 3.12.10.
pandas==3.0.5
streamlit==1.60.0
pydantic==2.13.4
pypdf==6.14.2

# starlette is a transitive dep of streamlit (its experimental ASGI/Starlette backend), but it's
# pinned explicitly here rather than left to resolve on its own: a later starlette release changed
# GZipResponder's signature to require a new keyword-only argument that streamlit==1.60.0 doesn't
# pass, breaking app startup entirely on any fresh install (verified on Streamlit Community Cloud)
# even though it works locally where an older starlette happened to already be installed.
starlette==1.3.1

# openpyxl is required at runtime, not just by the one-off synthetic-data generator scripts under
# data/synthetic/<company>/ - pandas needs it as the .xlsx engine for both normalize.py and
# app/streamlit_app.py's "Before" panel (pd.read_excel).
openpyxl==3.1.5

# reportlab is only used by data/synthetic/<company>/contracts/generate_contracts.py to build the
# synthetic contract PDFs - not needed to run the pipeline or the app.
reportlab==5.0.0
